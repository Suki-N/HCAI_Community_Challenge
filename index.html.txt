        // CLIFF CALCULATOR LOGIC
        function updateCliffCalc() {
            const wage = parseFloat(document.getElementById('input-wage').value);
            const family = parseInt(document.getElementById('input-family').value);
            const rent = parseFloat(document.getElementById('input-rent').value);

            document.getElementById('wage-val').innerText = `$${wage.toFixed(2)} / hr`;
            document.getElementById('family-val').innerText = `${family} Member${family > 1 ? 's' : ''}`;
            document.getElementById('rent-val').innerText = `$${rent.toLocaleString()} / mo`;

            const grossMonthly = wage * 40 * 4.33;
            document.getElementById('calc-income').innerText = `$${Math.round(grossMonthly).toLocaleString()}`;

            const fplBase = 15000 + (family - 1) * 5380;
            const fplPct = Math.round((grossMonthly * 12 / fplBase) * 100);
            document.getElementById('calc-fpl').innerText = `${fplPct}% Poverty Level`;

            let snapBenefit = 0;
            let snapEligible = true;
            if (fplPct <= 130) {
                snapBenefit = Math.max(50, Math.round(600 - (grossMonthly * 0.2)));
            } else {
                snapEligible = false;
            }

            if (snapEligible) {
                document.getElementById('calc-snap').innerText = `$${snapBenefit} / mo`;
                document.getElementById('calc-snap-status').innerText = "Eligible (<130% FPL)";
                document.getElementById('calc-snap-status').className = "text-[10px] text-emerald-700 font-semibold block mt-0.5";
            } else {
                document.getElementById('calc-snap').innerText = "$0 / mo";
                document.getElementById('calc-snap-status').innerText = "Ineligible (Cliff Cutoff)";
                document.getElementById('calc-snap-status').className = "text-[10px] text-rose-600 font-bold block mt-0.5";
            }

            const netDisposable = grossMonthly - rent;
            const rentPct = Math.round((rent / grossMonthly) * 100);

            document.getElementById('calc-net').innerText = `$${Math.round(netDisposable).toLocaleString()}`;
            document.getElementById('calc-rent-pct').innerText = `${rentPct}% Rent Burden`;

            const alertBox = document.getElementById('cliff-alert-box');
            if (!snapEligible) {
                alertBox.className = "mt-4 p-4 bg-rose-50 border border-rose-200 rounded-xl text-xs text-rose-900 leading-relaxed";
                alertBox.innerHTML = `<strong class="font-bold text-rose-900">CLIFF LOSS ACTIVE:</strong> At $${wage.toFixed(2)}/hr (${fplPct}% FPL), earnings exceed SNAP limits. Food assistance drops to zero, forcing the family to absorb a ${rentPct}% rent burden without food subsidies (Gaddy & Morrell 2026).`;
                document.getElementById('cliff-status-badge').className = "text-xs px-3 py-1 rounded-full bg-rose-100 text-rose-800 border border-rose-300 font-bold";
                document.getElementById('cliff-status-badge').innerText = "Benefit Loss Active";
            } else {
                alertBox.className = "mt-4 p-4 bg-amber-50 border border-amber-200 rounded-xl text-xs text-amber-900 leading-relaxed";
                alertBox.innerHTML = `<strong class="font-bold text-amber-900">Near-Cliff Threshold:</strong> Earning $${wage.toFixed(2)}/hr maintains $${snapBenefit}/mo in SNAP relief, but housing costs consume ${rentPct}% of gross earnings. Any raise past 130% FPL risks total loss of food benefits.`;
                document.getElementById('cliff-status-badge').className = "text-xs px-3 py-1 rounded-full bg-amber-100 text-amber-800 border border-amber-300 font-bold";
                document.getElementById('cliff-status-badge').innerText = "Within Safety Net";
            }
        }

        // RUN NARRATIVE ASSIST ANALYSIS
        async function runNarrativeAssist() {
            const profile = empiricalProfiles[currentClientId];
            const btn = document.getElementById('run-assist-btn');
            const summaryOutput = document.getElementById('ai-summary-output');
            const tagsContainer = document.getElementById('barrier-tags-container');

            btn.disabled = true;
            btn.innerHTML = `<svg class="animate-spin h-4 w-4 text-white" xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24"><circle class="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" stroke-width="4"></circle><path class="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"></path></svg> Synthesizing Note...`;

            if (geminiApiKey) {
                try {
                    const systemPrompt = "You are a Human-Centered AI assistant for Charlotte intake social workers. Summarize the intake note in 2 concise sentences emphasizing housing stability and food security needs. Cite relevant evidence.";
                    const userQuery = `Summarize this empirical case note: ${profile.rawText}`;
                    
                    const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${geminiApiKey}`, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify({
                            contents: [{ parts: [{ text: userQuery }] }],
                            systemInstruction: { parts: [{ text: systemPrompt }] }
                        })
                    });

                    const data = await response.json();
                    if (data.candidates && data.candidates[0].content.parts[0].text) {
                        summaryOutput.value = data.candidates[0].content.parts[0].text;
                    } else {
                        summaryOutput.value = profile.summary;
                    }
                } catch(e) {
                    summaryOutput.value = profile.summary;
                }
            } else {
                await new Promise(r => setTimeout(r, 500));
                summaryOutput.value = profile.summary;
            }

            // Render Barrier Tags with explainability linking
            tagsContainer.innerHTML = "";
            profile.barriers.forEach(b => {
                let badgeStyle = "bg-rose-100 text-rose-800 border-rose-300";
                if(b.type === "food") badgeStyle = "bg-amber-100 text-amber-800 border-amber-300";
                if(b.type === "health") badgeStyle = "bg-emerald-100 text-emerald-800 border-emerald-300";
                if(b.type === "income") badgeStyle = "bg-purple-100 text-purple-800 border-purple-300";

                const tagBtn = document.createElement('button');

                tagBtn.onclick = () => highlightSourceText(b.match, b.type);
                tagsContainer.appendChild(tagBtn);
            });

            renderHighlightedRawText(profile);

            btn.disabled = false;
            btn.innerHTML = `<svg class="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M19.428 15.428a2 2 0 00-1.022-.547l-2.387-.477a6 6 0 00-3.86.517l-.318.158a6 6 0 01-3.86.517L6.05 15.21a2 2 0 00-1.806.547M8 4h8l-1 1v5.172a2 2 0 00.586 1.414l5 5c1.26 1.26.367 3.414-1.415 3.414H4.828c-1.782 0-2.674-2.154-1.414-3.414l5-5A2 2 0 009 10.172V5L8 4z"></path></svg> <span>Run Narrative-Assist Analysis</span>`;
        }

        // EXPLAINABILITY HIGHLIGHTING ENGINE
        function renderHighlightedRawText(profile) {
            let html = profile.rawText;
            profile.barriers.forEach(b => {
                const highlightClass = `highlight-${b.type}`;
                const regex = new RegExp(`(${escapeRegExp(b.match)})`, 'gi');
                html = html.replace(regex, `<span class="${highlightClass}" onclick="showTagNotice('${b.label}')">$1</span>`);
            });
            document.getElementById('raw-notes-display').innerHTML = html;
        }

        function highlightSourceText(matchText, type) {
            const container = document.getElementById('raw-notes-display');

                    s.classList.add('highlight-active');
                    setTimeout(() => s.classList.remove('highlight-active'), 2000);
                }
            }
        }

        function approveCaseFile() {
            const summary = document.getElementById('ai-summary-output').value;
            if(!summary) {
                alertCustom("Please run Narrative-Assist analysis before finalizing referral.");
                return;
            }
            alertCustom(`Referral signed off by Caseworker! Exported to Charlotte Coordinated Entry System.`);
        }

        // CUSTOM MODAL NOTIFICATION
        function alertCustom(msg) {
            const toast = document.createElement('div');
            toast.className = "fixed bottom-6 right-6 bg-emerald-700 text-white text-xs font-bold px-4 py-3 rounded-xl shadow-2xl z-50 flex items-center gap-2 transition transform translate-y-0 border border-emerald-500";
            toast.innerHTML = `<svg class="w-5 h-5 text-emerald-200" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M5 13l4 4L19 7"></path></svg> <span>${msg}</span>`;
            document.body.appendChild(toast);
            setTimeout(() => toast.remove(), 4000);
        }

        function saveGeminiKey() {
            const key = document.getElementById('gemini-key-input').value.trim();
            if(key) {
                geminiApiKey = key;
                localStorage.setItem("gemini_api_key", key);
                document.getElementById('api-key-status').innerText = "Gemini API Active";
            }
            closeGeminiModal();
        }

        function clearGeminiKey() {
            geminiApiKey = "";
            localStorage.removeItem("gemini_api_key");
            document.getElementById('gemini-key-input').value = "";
            document.getElementById('api-key-status').innerText = "Gemini API Config";
            closeGeminiModal();
        }
